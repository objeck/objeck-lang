Libraries

64 libraries, generated from core/compiler/build_libs.sh and the sources. Pass them to obc -lib; a library's own dependencies are not added for you. API reference.

obc -src app.obs -lib json,net,cipher

In practice

JSON

use Data.JSON;

class Main {
  function : Main(args : String[]) ~ Nil {
    root := JsonParser->TextToElement("{\"name\": \"Ada\", \"langs\": [\"en\", \"fr\"], \"age\": 36}");
    name := root->Get("name")->GetString();
    age := root->Get("age")->GetInt();
    "{$name} is {$age}"->PrintLine();

    doc := JsonElement->New(JsonElement->JsonType->OBJECT);
    doc->Insert("ok", true);
    doc->Insert("count", 3);
    doc->ToString()->PrintLine();
  }
}
$ obc -src json.obs -lib json
$ obr json.obe
Ada is 36
{"count":3,"ok":true}

SQL

use Query.Structured;

class Main {
  function : Main(args : String[]) ~ Nil {
    sales := Table->New("sales", ["region", "qty"]);
    Add(sales, "east", 10);
    Add(sales, "west", 4);
    Add(sales, "east", 3);
    result := sales->Query("select region, sum(qty) as total from sales group by region");
    result->ToCsv()->Print();
  }

  function : Add(table : Table, region : String, qty : Int) ~ Nil {
    row := table->Insert();
    row->Set("region", region);
    row->Set("qty", IntRef->New(qty));
  }
}
$ obc -src query.obs -lib query,misc,net,json,regex,xml,csv,cipher
$ obr query.obe
region,total
east,13
west,4

Machine learning

use System.ML;

class Main {
  function : Main(args : String[]) ~ Nil {
    x := Float->New[5, 1];
    y := Float->New[5, 1];
    each(i : 5) {
      x[i, 0] := i->As(Float);
      y[i, 0] := 2.0 * i->As(Float) + 1.0;
    };

    model := LinearRegression->New();
    model->Fit(x, y, true);
    c := model->GetCoefficients();
    "y = {$(c[1]):.2} x + {$(c[0]):.2}"->PrintLine();
  }
}
$ obc -src ml.obs -lib csv,ml
$ obr ml.obe
y = 2.00 x + 1.00

Threads

use System.Concurrency;

class Counter from Thread {
  @total : static : Int;
  @lock : static : ThreadMutex;

  New() {
    Parent();
  }

  function : Init() ~ Nil {
    @lock := ThreadMutex->New("lock");
  }

  method : public : Run(arg : System.Base) ~ Nil {
    each(i : 1000) {
      critical(@lock) {
        @total += 1;
      };
    };
  }

  function : Total() ~ Int {
    return @total;
  }
}

class Main {
  function : Main(args : String[]) ~ Nil {
    Counter->Init();
    workers := Counter->New[4];
    each(i : workers) {
      workers[i] := Counter->New();
      workers[i]->Execute(Nil);
    };
    each(i : workers) {
      workers[i]->Join();
    };
    Counter->Total()->PrintLine();
  }
}
$ obc -src threads.obs -lib concurrent
$ obr threads.obe
4000
